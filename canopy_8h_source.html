1/* canopy.h — C ABI for the canopy piecewise-Chebyshev approximator.
2 *
3 * This surface mirrors the C++ `canopy::fit(f, a, b, tol, options)` API
4 * (see include/canopy/canopy.hpp): the domain is given as its lower/upper
5 * corners `a`/`b` and the fit knobs live in `canopy_opts`. It is *not* the
6 * legacy `canopy_input_t` / `canopy_init` shape.
7 *
8 * Precision lives in the library prefix, FINUFFT/FFTW style: the `canopy_*`
9 * entry points operate on `double`, the `canopyf_*` twins on `float`. Only
10 * the value-carrying functions (fit + eval family) come in twin form; the
11 * introspection and lifetime functions are dtype-independent and operate on
12 * the opaque handle, so they stay single (`canopy_` prefix, no `canopyf_`).
13 *
14 * Degree is baked: the leaf Chebyshev degree is fixed to 7 for all
15 * (arch, dtype, input_dim) cells (see arch_degree_table.hpp). It is not a
16 * user parameter.
17 *
18 * Eval-only: a fitted function can be evaluated (single, AoS batch,
19 * sorted-1D, SoA/transposed) but not serialized. There is one opaque handle
20 * type carrying a runtime dtype tag (f64 or f32).
21 *
22 * Thread safety: once `canopy_fit`/`canopyf_fit` returns a handle, its eval
23 * functions are safe to call concurrently from multiple threads provided
24 * each call writes a disjoint output slice (the underlying C++ Function
25 * allocates per-call scratch). `canopy_last_error()` is thread-local.
26 *
27 * No C++ exception ever crosses this boundary: fit failures return NULL
28 * and set `canopy_last_error()`.
29 */

52/* Fit knobs — mirrors canopy::options. `int` is used for the bool field so
53 * the struct is plain C. */
54typedef struct {
55 canopy_tol_kind_t tol_kind;
56 int max_depth;
57 /* Leaf-storage cap during the fit, in MiB. Tri-state: <0 (the default in
58 * canopy_default_opts) auto-selects a small, dimension-scaled budget
59 * (4/8/16 MiB for input_dim 1/2/3); 0 disables the cap; >0 is an explicit
60 * cap. Crossing it makes the fit return NULL with canopy_last_error()
61 * naming the offending panel. */
62 int max_memory_mib;
63 int allow_max_depth_leaves;
64 int min_uniform_depth;
65} canopy_opts;
