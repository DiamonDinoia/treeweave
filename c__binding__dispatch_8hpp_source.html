1// c_binding_dispatch.hpp — definition of the per-(arch, dtype, dim, degree)
2// factory `make_eval_for`.
3//
4// No include guard and no namespace of its own: each variant TU includes this
5// at `canopy::capi` scope (NOT inside the anonymous namespace) so the
6// instantiated `make_eval_for<Arch, …>` keeps *external* linkage, while its
7// body still names the anonymous-namespace `EvalFactory` (visible from the
8// enclosing namespace). The variant TU then explicitly instantiates exactly
9// one (best_arch, dtype, dim, degree) here.
10//
11// This is the single site of the output_dim dispatch: an O(1) contiguous poet
12// table over {1, 2, 3}. `Arch` is passed to `EvalFactory<Arch,...>` so that
13// every poly_eval template instantiation inside EvalImpl carries `Arch` as a
14// phantom type, giving each `-march` variant a distinct mangled COMDAT key
15// and preventing the linker from folding the four per-arch kernel copies.
16
17template <class Arch, class T, std::size_t IN, int Deg>
18auto make_eval_for(int output_dim, c_func_t<T> f, void *data, const T *a, const T *b, double tol,
19 const canopy::options &opts) -> IEval<T> * {
20 return poet::dispatch(EvalFactory<Arch, T, IN, Deg>{f, data, a, b, tol, opts},
21 poet::DispatchParam<poet::make_range<1, 3>>{output_dim});
22}
