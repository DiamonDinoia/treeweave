54 auto fit(const TreeInput &input, const Func &func, const Value<value_type, input_dim> &center,
55 const Value<value_type, input_dim> &half_length, const std::vector<value_type> &samples,
56 std::vector<poly_eval_type> &polyfits) -> bool {
57 if (!samples.empty())
58 throw std::runtime_error("Canopy fit error: sample points not yet supported");
59
60 const auto n_polyfit_before = polyfits.size();
61 const input_type lb = center - half_length;
62 const input_type ub = center + half_length;
63
64 auto rollback_and_fail = [&polyfits, n_polyfit_before]() -> bool {
65 while (polyfits.size() != n_polyfit_before)
66 polyfits.pop_back();
67 return false;
68 };
69
70 auto polyfit = polyfits.emplace_back(func, lb, ub);
71
72 // tail_error reads coefficients out of polyfit's 1D `FuncEval` and is
73 // not meaningful for the ND / array-output path (which uses
74 // `FuncEvalND` with multi-axis coefficient storage). Gate the call so
75 // array/ND fits still compile, and fail loudly at runtime if the user
76 // asks for a Tail TolKind on an unsupported shape.
77 constexpr bool kTailErrorSupported =
78 !poly_eval::detail::hasTupleSize_v<input_type> && !poly_eval::detail::hasTupleSize_v<output_type>;
79 const bool wants_tail = input.tol_kind == TolKind::RelativeTail || input.tol_kind == TolKind::AbsoluteTail;
80 if constexpr (kTailErrorSupported) {
81 if (wants_tail) {
82 if (tail_error_below_tolerance(input.tol_kind, input.tol, polyfit))
83 return rollback_and_fail();
84 } else if (sample_error_below_tolerance(kFitSamplesPerDim, input.tol_kind, input.tol, center, half_length,
85 func, polyfit)) {
86 return rollback_and_fail();
87 }
88 } else {
89 if (wants_tail)
90 throw std::runtime_error("Canopy fit error: TolKind::RelativeTail / AbsoluteTail "
91 "is only supported for 1D scalar→scalar fits; use a "
92 "sample-based TolKind for array-valued or ND fits");
93 if (sample_error_below_tolerance(kFitSamplesPerDim, input.tol_kind, input.tol, center, half_length, func,
94 polyfit))
95 return rollback_and_fail();
96 }
97
98 poly_eval_id_ = static_cast<std::uint32_t>(n_polyfit_before);
99 return true;
100 }

104 void force_fit_as_leaf(const Func &func, const Value<value_type, input_dim> &center,
105 const Value<value_type, input_dim> &half_length, std::vector<poly_eval_type> &polyfits) {
106 const input_type lb = center - half_length;
107 const input_type ub = center + half_length;
108 polyfits.emplace_back(func, lb, ub);
109 poly_eval_id_ = static_cast<std::uint32_t>(polyfits.size() - 1);
110 }
