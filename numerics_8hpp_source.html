22template <int EXP, typename T>
23constexpr auto powi(T base) -> T {
24 if constexpr (EXP == 0) {
25 return T{1};
26 } else if constexpr (EXP % 2 == 0) {
27 const auto half = powi<EXP / 2>(base);
28 return half * half;
29 } else {
30 return base * powi<EXP - 1>(base);
31 }
32}

62template <class Polyfit>
63auto tail_error_below_tolerance(TolKind tol_type, double tol, const Polyfit &polyfit) -> bool {
64 constexpr std::size_t input_dim = value_dim_v<typename Polyfit::InputType>;
65 constexpr std::size_t output_dim = value_dim_v<typename Polyfit::OutputType>;
66 using T = poly_eval::detail::value_type_or_t<typename Polyfit::InputType>;
67 // ND / array-output never reaches here: the sole caller (node.hpp) gates
68 // this behind `if constexpr (kTailErrorSupported)`, which is true only for
69 // scalar→scalar fits. The static_asserts document that contract and turn
70 // any stray ND instantiation into a compile error rather than dead code.
71 static_assert(input_dim == 1, "tail_error check is only implemented for 1D scalar input; "
72 "use a sample-based tol_type for array-valued or ND fits");
73 static_assert(output_dim == 1, "tail_error only implemented for single output in 1D");
74
75 T maxcoeff{0.0};
76 T scaling_factor{1.0};
77
78 const auto &coeffs = polyfit.coeffs();
79 constexpr std::size_t N = Polyfit::NCOEFFS;
80 for (std::size_t i = 0; i < 2; ++i)
81 maxcoeff = std::max(std::abs(coeffs[i]), maxcoeff);
82 scaling_factor = std::max(scaling_factor, std::abs(coeffs[N - 1]));
83
84 if (tol_type == TolKind::RelativeL2 || tol_type == TolKind::RelativeMax)
85 return static_cast<double>(maxcoeff / scaling_factor) > tol;
86 else
87 return static_cast<double>(maxcoeff) > tol;
88}

93template <class Func, class Polyfit>
94inline auto sample_error_below_tolerance(int n_sample_1d, TolKind tol_type, double tol,
95 const typename Polyfit::InputType &center_in,
96 const typename Polyfit::InputType &half_length_in, const Func &func,
97 const Polyfit &polyfit) -> bool {
98 constexpr std::size_t input_dim = value_dim_v<typename Polyfit::InputType>;
99 constexpr std::size_t output_dim = value_dim_v<typename Polyfit::OutputType>;
100 // Sample in the fit's own value type so a `float` fit never silently
101 // promotes its sample grid to double; the error metrics below still
102 // accumulate in double via explicit casts.
103 using T = poly_eval::detail::value_type_or_t<typename Polyfit::InputType>;
104 const auto n_sample_1d_sz = static_cast<std::size_t>(n_sample_1d);
105 const std::size_t n_samples = powi<static_cast<int>(input_dim)>(n_sample_1d_sz);
106 const Value half_length = half_length_in;
107 const Value center = center_in;
108
109 double max_abs_err{0.0};
110 double max_rel_err{0.0};
111 double abs_err_l2{0.0};
112 double direct_sum{0.0};
113 for (std::size_t linear_index = 0; linear_index < n_samples; ++linear_index) {
114 Value<T, input_dim> sample_point;
115 std::size_t curr_index = linear_index;
116
117 for (std::size_t dim = 0; dim < input_dim; ++dim) {
118 const T dx = T{2} * half_length[dim] / static_cast<T>(n_sample_1d_sz);
119 sample_point[dim] =
120 center[dim] - half_length[dim] + dx / T{2} + dx * static_cast<T>(curr_index % n_sample_1d_sz);
121 curr_index /= n_sample_1d_sz;
122 }
123
124 Value<T, output_dim> actual = func(sample_point);
125 Value<T, output_dim> approx = polyfit(sample_point);
126
127 for (std::size_t i = 0; i < output_dim; ++i) {
128 const double abs_err = std::abs(static_cast<double>(approx[i]) - static_cast<double>(actual[i]));
129 max_abs_err = std::max(max_abs_err, abs_err);
130 if (actual[i] != T{0})
131 max_rel_err = std::max(max_rel_err, std::abs(abs_err / static_cast<double>(actual[i])));
132 abs_err_l2 += powi<2>(abs_err);
133 direct_sum += powi<2>(static_cast<double>(actual[i]));
134 }
135 }
136
137 switch (tol_type) {
138 case TolKind::RelativeL2:
139 return std::sqrt(abs_err_l2 / direct_sum) > tol;
140 case TolKind::AbsoluteL2:
141 return std::sqrt(abs_err_l2) / static_cast<double>(n_samples * output_dim) > tol;
142 case TolKind::RelativeMax:
143 return max_rel_err > tol;
144 case TolKind::AbsoluteMax:
145 return max_abs_err > tol;
146 default:
147 throw std::runtime_error("Canopy fit error: unknown tolerance type for sampling");
148 }
149}
