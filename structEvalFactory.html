template<class Arch, class T, std::size_t IN, int Deg>
struct EvalFactory< Arch, T, IN, Deg >

poet::dispatch functor for the output_dim switch within one fixed (arch, value_type, input_dim, degree). operator()<Out>() builds the concrete EvalImpl for the matched output_dim; poet returns nullptr for any output_dim outside the instantiated range.
