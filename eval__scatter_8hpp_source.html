48template <std::size_t Degree, class Func>
49auto eval_scatter_sorted(std::span<const Function<Degree, Func> *const> fits, std::span<const std::uint32_t> fit_ids,
50 std::span<const double> xs, std::span<double> ys, std::uint32_t n_fits,
51 std::span<std::uint32_t> counts, std::span<std::uint32_t> perm) -> void {
52 using fn_t = Function<Degree, Func>;
53 static_assert(fn_t::input_dim == 1 && fn_t::output_dim == 1, "eval_scatter_sorted: 1D scalar fits only");
54
55 const std::size_t n = fit_ids.size();
56 assert(xs.size() == n && ys.size() == n);
57 assert(counts.size() >= n_fits && "counts span too small");
58 assert(perm.size() >= n && "perm span too small");
59 if (n == 0)
60 return;
61
62 for (std::size_t i = 0; i < n; ++i) {
63 assert(fit_ids[i] < n_fits && "fit_ids[i] >= n_fits");
64 ++counts[fit_ids[i]];
65 }
66
67 // Reinecke's trick: in-place exclusive scan turns counts into start
68 // offsets. The placement loop below bumps each entry, so after
69 // placement counts[k] holds the END of bin k.
70 std::exclusive_scan(counts.begin(), counts.begin() + n_fits, counts.begin(), std::uint32_t{0});
71
72 // Placement: write only the original-order index at the bin cursor.
73 // The dispatch loop reads xs[perm[k]] directly, so no fit-grouped
74 // `xs_sorted` scratch is needed — trades sequential xs_sorted reads
75 // for scattered reads of xs, but dispatch already writes ys[perm[k]]
76 // scattered, so the cache pattern is the same gather/scatter pair.
77 for (std::size_t i = 0; i < n; ++i) {
78 const std::uint32_t fid = fit_ids[i];
79 perm[counts[fid]++] = static_cast<std::uint32_t>(i);
80 }
81
82 // Dispatch each non-empty fit-id run. fn(x) is CANOPY_ALWAYS_INLINE,
83 // so the per-pair body is an inlined Hybrid FMA chain with no callq.
84 std::uint32_t run_start = 0;
85 for (std::uint32_t fid = 0; fid < n_fits; ++fid) {
86 const std::uint32_t run_end = counts[fid];
87 const auto &fn = *fits[fid];
88 for (std::uint32_t k = run_start; k < run_end; ++k) {
89 const std::uint32_t idx = perm[k];
90 ys[idx] = fn(xs[idx]);
91 }
92 run_start = run_end;
93 }
94}

99template <std::size_t Degree, class Func>
100auto eval_scatter_sorted(std::span<const Function<Degree, Func> *const> fits, std::span<const std::uint32_t> fit_ids,
101 std::span<const double> xs, std::span<double> ys, std::uint32_t n_fits) -> void {
102 std::vector<std::uint32_t> counts(n_fits);
103 std::vector<std::uint32_t> perm(fit_ids.size());
104 eval_scatter_sorted<Degree, Func>(fits, fit_ids, xs, ys, n_fits, std::span<std::uint32_t>(counts),
105 std::span<std::uint32_t>(perm));
106}
