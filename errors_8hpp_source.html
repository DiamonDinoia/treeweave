27inline auto format_domain(std::ostringstream &os, const std::vector<double> &a, const std::vector<double> &b) -> void {
28 if (a.size() == 1) {

43class MaxDepthExceeded : public std::exception {
44 public:
45 MaxDepthExceeded() = default;
46
47 template <class VecC, class VecH>
48 MaxDepthExceeded(std::size_t depth, const VecC &center_in, const VecH &half_length_in) : depth_(depth) {
49 auto cit = center_in.begin();
50 auto hit = half_length_in.begin();
51 for (; cit != center_in.end() && hit != half_length_in.end(); ++cit, ++hit) {
52 // Panel bounds are reported in double regardless of the fit's
53 // value_type; cast explicitly so a float fit does not trip
54 // -Wdouble-promotion under -Werror.
55 a_.push_back(static_cast<double>(*cit - *hit));
56 b_.push_back(static_cast<double>(*cit + *hit));
57 }
58 std::ostringstream os;
59 os << "Canopy fit error: tree depth exceeded max allowed input depth (" << depth_ << ") on panel ";
60 detail::format_domain(os, a_, b_);
61 os << " — likely a singularity in this interval; "
62 "subdivide manually away from the singularity or raise options.max_depth.";
63 msg_ = os.str();
64 }
65
69 explicit MaxDepthExceeded(std::vector<NonConvergedPanel> panels) : panels_(std::move(panels)) {
70 if (!panels_.empty()) {
71 depth_ = panels_.front().depth;
72 a_ = panels_.front().a;
73 b_ = panels_.front().b;
74 }
75 std::ostringstream os;
76 os << "Canopy fit error: tree depth exceeded max allowed input depth (" << depth_ << ") on " << panels_.size()
77 << " panel" << (panels_.size() == 1 ? "" : "s") << "; first ";
78 detail::format_domain(os, a_, b_);
79 os << " — likely a singularity; subdivide manually, raise "
80 "options.max_depth, or set options.allow_max_depth_leaves=true "
81 "to accept best-effort leaves.";
82 msg_ = os.str();
83 }

88 [[nodiscard]] auto a() const noexcept -> const std::vector<double> & { return a_; }
90 [[nodiscard]] auto b() const noexcept -> const std::vector<double> & { return b_; }
92 [[nodiscard]] auto panels() const noexcept -> const std::vector<NonConvergedPanel> & { return panels_; }
93 [[nodiscard]] auto what() const noexcept -> const char * override { return msg_.c_str(); }
94
95 private:
96 std::size_t depth_ = 0;
97 std::vector<double> a_;
98 std::vector<double> b_;
99 std::vector<NonConvergedPanel> panels_;

101};
